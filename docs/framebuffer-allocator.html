중복 할당과 내보내기 실패를 구분하기

FrameBufferAllocator::allocate에서 try_emplace의 inserted가 false이면 오류를 기록하고 -EBUSY를 반환합니다. FrameBufferAllocator::allocate()는 exportFrameBuffers()의 반환값을 변수 ret에 저장하고, ret가 음수이면 buffers_에서 해당 항목을 erase로 제거합니다. FrameBufferAllocator::allocate는 exportFrameBuffers 호출 후 ret의 부호를 검사하는 if문 밖에서 ret의 부호와 관계없이 ret을 반환합니다.

int FrameBufferAllocator::allocate(Stream *stream)
{
    const auto &[it, inserted] = buffers_.try_emplace(stream);

    if (!inserted) {
        LOG(Allocator, Error) << "Buffers already allocated for stream";
        return -EBUSY;
    }

    int ret = camera_->exportFrameBuffers(stream, &it->second);
    if (ret == -EINVAL)
        LOG(Allocator, Error)
            << "Stream is not part of " << camera_->id()
            << " active configuration";

    if (ret < 0)
        buffers_.erase(it);

    return ret;
}

int FrameBufferAllocator::free(Stream *stream)
{
    auto iter = buffers_.find(stream);
    if (iter == buffers_.end())
        return -EINVAL;

    buffers_.erase(iter);

    return 0;
}
